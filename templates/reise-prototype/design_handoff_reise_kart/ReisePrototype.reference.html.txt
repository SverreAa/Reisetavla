      named(pts).forEach(k => { const big = k === 0 || k === last; dots.push({ x: pts[k].x, y: pts[k].y, r: big ? 6 : 3.5, c: mc(l.mode), sw: big ? 3 : 2, op: passedAt(k) && !(k === s.i && li === s.li) ? 0.4 : 1 }); });
      const legDone = li < s.li || (li === s.li && s.i === last);
      const f = l.flight ? FLIGHTS[l.flight] : null;
      labels.push({ x: pts[0].x, y: pts[0].y, side: l.lab[0], c: mc(l.mode), op: legDone ? 0.55 : 1, kind: 'På · ' + MODE[l.mode] + ' ' + l.code, name: pts[0].n, meta: l.start + ' · ' + (f ? 'Gate ' + f.gate : l.plat) });
      labels.push({ x: pts[last].x, y: pts[last].y, side: l.lab[1], c: mc(l.mode), op: legDone ? 0.55 : 1, kind: 'Av · ' + MODE[l.mode] + ' ' + l.code, name: pts[last].n, meta: l.end + (f ? ' · bånd ' + f.belt : '') });
      if (showControl && l.control) { const cp = pts.find(p => p.n === l.control.at); if (cp) controls.push({ x: cp.x, y: cp.y }); }
    });
    const curLeg = t.legs[s.li], curPts = legPts(curLeg), cur = curPts[s.i];
    let alertTitle = '', alertText = '';
    if (this.state.alertOn) {
      if (curLeg.mode !== 'foot' && s.i === curPts.length - 1) { alertTitle = 'Gå av nå'; alertText = `${cur.n} · ${MODE[curLeg.mode]} ${curLeg.code}`; }
      else if (ns) { const nl = t.legs[ns.li], np = legPts(nl); if (nl.mode !== 'foot' && ns.i === np.length - 1) { alertTitle = 'Gå av ved neste stopp'; alertText = `${np[ns.i].n} · ankomst ${nl.end}`; } }
    }
    const onBoard = curLeg.mode !== 'foot' && s.i < curPts.length - 1;
    let status = onBoard ? `Om bord på ${MODE[curLeg.mode].toLowerCase()} ${curLeg.code} · ved ${cur.n}` : `Du er ved ${cur.n}`;
    if (ns) status += ` · neste: ${legPts(t.legs[ns.li])[ns.i].n}`;
    else { const air = t.legs.find(l => l.flight); if (air) status += ` · gate ${FLIGHTS[air.flight].gate} åpner ${FLIGHTS[air.flight].gateOpen}`; }

    let vehicles = [], groups = [], routeTitle = '';
    if (route) {
      const leg = t.legs.find(l => l.route === this.state.mapRoute);
      const my = leg.fi, myName = route.pts[my].n;
      const nm = named(route.pts);
      routeTitle = `${MODE[route.mode]} ${route.code} i trafikk nå`;
      const list = this.vp[this.state.mapRoute].map(([p, d]) => {
        const i = Math.min(Math.floor(p), route.pts.length - 2), f = p - i, A = route.pts[i], B = route.pts[i + 1];
        const toward = (d > 0 && p < my) || (d < 0 && p > my);
        const near = route.pts[nm.slice().sort((a, b) => Math.abs(a - p) - Math.abs(b - p))[0]].n;
        const min = Math.max(1, Math.round(Math.abs(my - p) * 1.6));
        return { x: A.x + (B.x - A.x) * f, y: A.y + (B.y - A.y) * f, code: route.code, bg: mc(route.mode), op: toward ? 1 : 0.5, ring: toward ? '0 0 0 3px var(--surface),0 0 0 5px var(--accent)' : '0 0 0 3px var(--surface)', d, toward, dist: Math.abs(my - p), where: `Ved ${near}`, status: toward ? `om ${min} min til ${myName}` : `har passert ${myName}`, away: !toward };
      });
      vehicles = list;
      groups = [1, -1].map(d => ({ title: 'Mot ' + route.ends[d > 0 ? 1 : 0], note: d > 0 ? 'Din retning' : 'Motsatt spor', rows: list.filter(v => v.d === d).sort((a, b) => (b.toward - a.toward) || (a.dist - b.dist)) }));
    }
    const ctrlLeg = showControl ? t.legs.find(l => l.control) : null;
    const svg = h('svg', { width: 390, height: 844, style: { position: 'absolute', inset: 0 } },
      segs.map((g, i) => h('polyline', { key: 's' + i, points: g.pts, fill: 'none', stroke: g.color, strokeWidth: g.w, strokeOpacity: g.op, strokeDasharray: g.dash || undefined, strokeLinecap: 'round', strokeLinejoin: 'round' })),
      dots.map((d, i) => h('circle', { key: 'd' + i, cx: d.x, cy: d.y, r: d.r, fill: 'var(--surface)', stroke: d.c, strokeWidth: d.sw, opacity: d.op })));
    const BOUND = { l: 6, r: 384, t: 152, b: 458 };
    const boxAt = (side, x, y, w, hh) => ({ tl: [x - 8 - w, y - 8 - hh], tr: [x + 8, y - 8 - hh], bl: [x - 8 - w, y + 8], br: [x + 8, y + 8], l: [x - 12 - w, y - hh / 2], r: [x + 12, y - hh / 2], t: [x - w / 2, y - 12 - hh], b: [x - w / 2, y + 12] })[side];
    const placed = [];
    const hit = (a, b) => a.x < b.x + b.w + 4 && b.x < a.x + a.w + 4 && a.y < b.y + b.h + 4 && b.y < a.y + a.h + 4;
    const fits = bx => bx.x >= BOUND.l && bx.x + bx.w <= BOUND.r && bx.y >= BOUND.t && bx.y + bx.h <= BOUND.b && !placed.some(p => hit(p, bx));
    const ovA = (a, b) => Math.max(0, Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x)) * Math.max(0, Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y));
    const obstacles = [];
    const sample = pts => { for (let k = 0; k < pts.length - 1; k++) { const A = pts[k], B = pts[k + 1], n = Math.max(1, Math.ceil(Math.hypot(B.x - A.x, B.y - A.y) / 10)); for (let j = 0; j <= n; j++) obstacles.push({ x: A.x + (B.x - A.x) * j / n - 5, y: A.y + (B.y - A.y) * j / n - 5, w: 10, h: 10, wt: 1 }); } };
    t.legs.forEach(l => { if (l.mode !== 'air') sample(legPts(l)); });
    obstacles.push({ x: cur.x - 16, y: cur.y - 16, w: 32, h: 32, wt: 50 });
    const place = (pref, x, y, w, hh) => {
      const order = [pref, 'tr', 'br', 'tl', 'bl', 'r', 'l', 't', 'b'].filter((s2, i, a) => a.indexOf(s2) === i);
      let best = null, bestScore = Infinity;
      order.forEach((sd, idx) => { const [bx, by] = boxAt(sd, x, y, w, hh); const bb = { x: bx, y: by, w, h: hh };
        const oob = Math.max(0, BOUND.l - bx) + Math.max(0, bx + w - BOUND.r) + Math.max(0, BOUND.t - by) + Math.max(0, by + hh - BOUND.b);
        const score = oob * 1e5 + placed.reduce((s3, p) => s3 + ovA(p, bb) * 100, 0) + obstacles.reduce((s3, o) => s3 + ovA(o, bb) * o.wt, 0) + idx;
        if (score < bestScore) { bestScore = score; best = bb; } });
      placed.push(best); return best; };
    for (let i = labels.length - 1; i > 0; i--) { const a = labels[i - 1], b = labels[i]; if (a.kind.startsWith('Av') && b.kind.startsWith('På') && a.name === b.name) { a.kind = 'Bytte · ' + b.kind.replace('På · ', ''); a.meta = a.meta.split(' · ')[0] + ' → ' + b.meta.split(' · ')[0]; a.c = b.c; labels.splice(i, 1); } }
    labels.forEach(l => { const w = Math.ceil(Math.max(l.kind.length * 7.4 + 13, l.name.length * 7.7, l.meta.length * 6.6)) + 20; const b = place(l.side, l.x, l.y, w, 54); l.bx = b.x; l.by = b.y; l.w = w; });
    controls.forEach(c => { const b = place('t', c.x, c.y, 28, 28); c.bx = b.x; c.by = b.y; });
    const lab = { position: 'absolute', background: 'var(--surface)', borderRadius: 10, boxShadow: 'var(--shadow)', padding: '5px 9px 6px', pointerEvents: 'none' };
    const overlay = h(React.Fragment, null,
      controls.map((c, i) => h('div', { key: 'c' + i, style: { position: 'absolute', left: c.bx, top: c.by, width: 28, height: 28, boxSizing: 'border-box', display: 'grid', placeItems: 'center', background: 'color-mix(in srgb,var(--warn) 14%,var(--surface))', color: 'var(--warn)', borderRadius: 999, boxShadow: 'var(--shadow)' }, 'aria-label': 'Billettkontroll meldt', title: 'Billettkontroll meldt' }, h(window.Reisetavla.Icon, { name: 'warn' }))),
      labels.map((l, i) => h('div', { key: 'l' + i, style: Object.assign({}, lab, { left: l.bx, top: l.by, width: l.w, height: 54, boxSizing: 'border-box', whiteSpace: 'nowrap', overflow: 'hidden', opacity: l.op }) },
        h('div', { style: { display: 'flex', alignItems: 'center', gap: 5, fontSize: 10.5, fontWeight: 700, letterSpacing: '.08em', textTransform: 'uppercase', color: 'var(--muted)', whiteSpace: 'nowrap' } }, h('i', { style: { width: 8, height: 8, borderRadius: '50%', flex: 'none', background: l.c } }), l.kind),
        h('div', { style: { fontWeight: 700, fontSize: 13.5, lineHeight: 1.2 } }, l.name),
        h('div', { style: { fontSize: 12, color: 'var(--muted)', fontVariantNumeric: 'tabular-nums', whiteSpace: 'nowrap' } }, l.meta))),
      vehicles.map((v, i) => h('div', { key: 'v' + i, style: { position: 'absolute', left: v.x, top: v.y, transform: 'translate(-50%,-50%)', width: 28, height: 28, borderRadius: '50%', background: v.bg, color: 'var(--on-mode)', display: 'grid', placeItems: 'center', fontFamily: 'var(--font-display)', fontWeight: 800, fontSize: 11, opacity: v.op, boxShadow: v.ring, transition: 'left 1s linear, top 1s linear' } }, v.code)),
      h('div', { 'aria-label': 'Din posisjon', style: { position: 'absolute', left: cur.x, top: cur.y, transform: 'translate(-50%,-50%)', width: 20, height: 20, borderRadius: '50%', background: 'var(--accent)', boxShadow: '0 0 0 4px var(--surface),0 0 0 10px color-mix(in srgb,var(--accent) 22%,transparent)', transition: 'left .4s, top .4s' } }));
    return {
      svg, overlay, trip: t, title: 'Pilestredet 35 → Bergen', labels, controls, vehicles, groups, routeTitle, hasRoute: !!route, me: { x: cur.x, y: cur.y },
      hasAlert: !!alertTitle, alertTitle, alertText, status,
      hasControl: !!ctrlLeg, control: ctrlLeg ? `Billettkontroll meldt ved ${ctrlLeg.control.at} på ${MODE[ctrlLeg.mode].toLowerCase()} ${ctrlLeg.code} for ${ctrlLeg.control.ago} min siden` : '',
      chips: t.legs.filter(l => l.route).map(l => { const on = this.state.mapRoute === l.route; return { code: l.code, mode: l.mode, label: on ? 'Vises' : ALL[l.mode], pressed: on, unpressed: !on, ring: on ? '0 0 0 2px var(--accent)' : 'var(--shadow)', onClick: () => this.setState(s2 => ({ mapRoute: s2.mapRoute === l.route ? null : l.route, sheetOpen: true })) }; }),
      legs: t.legs.map((l, li) => { const pts = legPts(l), done = li < s.li || (li === s.li && s.i === pts.length - 1 && l.mode !== 'air'); return { isFoot: l.mode === 'foot', isLine: l.mode !== 'foot', minutes: l.min, mode: l.mode, code: l.code, done, ahead: !done, line: l.mode === 'foot' ? `Gå ${l.m} m til ${pts[pts.length - 1].n}` : `${pts[0].n} → ${l.to || pts[pts.length - 1].n}`, meta: `${l.start}–${l.end}` + (l.crowd ? ' · ' + CROWD[l.crowd].label : '') + (l.flight ? ' · ' + FLIGHTS[l.flight].seats + ' ledige seter' : '') + (done ? ' · tilbakelagt' : '') }; })
    };
  }

  renderVals() {
    const s = this.state;
    const fr = this.filtered();
    const trips = fr.list.map(t => this.tripVM(t, fr.outside));
    const isMap = s.screen === 'map';
    const map = isMap ? this.mapVM() : null;
    const exclNames = EXCL.filter(e => s.excluded.includes(e.code)).map(e => e.code);
    let outsideText = '', raiseLabel = '', raiseTo = null;
    if (fr.outside) {
      const wl = walkLeg(fr.best), need = walkMax(fr.best);
      outsideText = `Ingen stopp innenfor ${fmtM(s.maxWalk)} gange. Nærmeste alternativ er ${wl.pts[1].n}, ${wl.m} m unna. Vi viser det beste forslaget utenfor maks gange.`;
      const opt = WALKS.find(w => Number(w.id) >= need); raiseTo = opt ? Number(opt.id) : null; raiseLabel = opt ? `Øk til ${opt.label}` : '';
    }

    return {
      theme: this.props.theme === 'light' || this.props.theme === 'dark' ? this.props.theme : undefined,

      routeRows: EXCL.map(e => { const ex = s.excluded.includes(e.code), w = sw(ex); return Object.assign({}, e, { excluded: ex, included: !ex, stateText: ex ? 'Ekskludert' : 'Med', aria: 'Ekskluder ' + e.name, track: w.track, knob: w.knob, toggle: () => this.setState(st => ({ excluded: st.excluded.includes(e.code) ? st.excluded.filter(c => c !== e.code) : st.excluded.concat(e.code) })) }); }),
      applyLabel: trips.length ? `Vis ${trips.length} ${trips.length === 1 ? 'reise' : 'reiser'}` : 'Lukk',
      fbCats: FB_CATS.map(c => ({ label: c, pressed: s.fbCat === c, unpressed: s.fbCat !== c, bg: s.fbCat === c ? 'var(--ink)' : 'var(--surface-2)', fg: s.fbCat === c ? 'var(--bg)' : 'var(--ink)', onClick: () => this.setState({ fbCat: c }) })),
      fbText: s.fbText, onFbText: e => this.setState({ fbText: e.target.value }),
      fbAttach: s.fbAttach, fbNoAttach: !s.fbAttach, attachTrack: sw(s.fbAttach).track, attachKnob: sw(s.fbAttach).knob, toggleAttach: () => this.setState(st => ({ fbAttach: !st.fbAttach })),
      sendFeedback: () => this.setState({ sheet: null, fbText: '', toast: 'Tilbakemeldingen er sendt' }),
      hasToast: !!s.toast, toast: s.toast,
      map: map || { labels: [], controls: [], vehicles: [], chips: [], legs: [], groups: [], me: { x: 0, y: 0 }, trip: {} }, mapSvg: map ? map.svg : null, mapOverlay: map ? map.overlay : null,
      goBack: () => this.setState({ screen: 'results' }),
      sheetOpen: s.sheetOpen, sheetClosed: !s.sheetOpen, sheetToggleLabel: s.sheetOpen ? 'Skjul detaljer' : 'Vis detaljer', toggleSheet: () => this.setState(st => ({ sheetOpen: !st.sheetOpen })),
      alertOn: s.alertOn, alertTrack: sw(s.alertOn).track, alertKnob: sw(s.alertOn).knob, toggleAlert: () => this.setState(st => ({ alertOn: !st.alertOn })),
      nextStep: () => this.setState(st => ({ step: Math.min(st.step + 1, steps(TRIPS.find(x => x.id === st.mapTrip) || TRIPS[1]).length - 1) })),
      resetStep: () => this.setState({ step: 0 })
    };
  }
